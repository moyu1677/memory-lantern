# 路由与存储代码摘录

这份文件集中展示聊天关怀报告接入网页时所需的原项目代码。它是**阅读用摘录**，不是可单独运行的新程序；正式代码仍在项目根目录的 `app.py` 和 `database.py`。

## 1. 数据表

```sql
CREATE TABLE IF NOT EXISTS care_reports (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    elder_id INTEGER NOT NULL,
    source_session_id TEXT NOT NULL,
    status TEXT NOT NULL
        CHECK (status IN ('signals_present', 'insufficient_or_conflicting')),
    summary TEXT NOT NULL,
    signals_json TEXT NOT NULL,
    created_at TEXT NOT NULL,
    FOREIGN KEY (elder_id) REFERENCES users(id) ON DELETE CASCADE
);

CREATE INDEX IF NOT EXISTS idx_care_reports_elder_created
ON care_reports(elder_id, created_at DESC);
```

不在这张表里存聊天原文，只保存经过校验的报告概括与信号。

## 2. 保存与家人读取

```python
def save_care_report(elder_id, source_session_id, report):
    connection = get_db()
    cursor = connection.execute(
        """
        INSERT INTO care_reports (
            elder_id, source_session_id, status, summary, signals_json, created_at
        ) VALUES (?, ?, ?, ?, ?, ?)
        """,
        (
            elder_id,
            source_session_id,
            report["status"],
            report["summary"],
            json.dumps(report["signals"], ensure_ascii=False),
            utc_now(),
        ),
    )
    connection.commit()
    return cursor.lastrowid


def list_care_reports(caregiver_id, limit=12):
    rows = get_db().execute(
        """
        SELECT report.id, report.elder_id, report.status, report.summary,
               report.signals_json, report.created_at,
               elder.display_name AS elder_name
        FROM care_reports AS report
        JOIN care_bindings AS binding ON binding.elder_id = report.elder_id
        JOIN users AS elder ON elder.id = report.elder_id
        WHERE binding.caregiver_id = ?
        ORDER BY report.created_at DESC
        LIMIT ?
        """,
        (caregiver_id, limit),
    ).fetchall()
    reports = []
    for row in rows:
        try:
            signals = json.loads(row["signals_json"])
        except (TypeError, json.JSONDecodeError):
            signals = []
        public_signals = []
        if isinstance(signals, list):
            for signal in signals:
                if isinstance(signal, dict):
                    public_signals.append({
                        key: signal.get(key, "")
                        for key in ("area", "topic", "strength", "trend", "family_action")
                    })
        reports.append({**dict(row), "signals": public_signals})
    return reports
```

关键点：`source_turn_ids` 被主动移除，家人页面没有办法拿到内部来源轮次。

```python
@app.route("/family-briefs/<int:elder_id>/generate", methods=["POST"])
@family_required
def generate_family_brief(elder_id):
    caregiver_id = g.current_user["id"]
    if not is_elder_bound_to(caregiver_id, elder_id):
        abort(403)

    # 家人只能总结老人已经完成的聊天，不能创建一段新聊天来生成报告。
    state = VOICE_CHAT_SESSIONS.get(elder_id)
    history = state.get("history", []) if state else []
    if not history:
        flash("这位老人最近还没有可用于生成报告的聊天内容。", "error")
        return redirect(url_for("family_briefs"))

    try:
        report = generate_care_report(deepseek_client, DEEPSEEK_MODEL, history)
    except Exception:
        app.logger.exception("家庭关怀报告生成失败：elder_id=%s", elder_id)
        flash("报告暂时生成失败，请稍后再试。", "error")
        return redirect(url_for("family_briefs"))

    if not report:
        flash("本次聊天内容还不足以形成可靠的关怀提示，请在更多自然聊天后再生成。", "error")
        return redirect(url_for("family_briefs"))

    save_care_report(elder_id, state.get("session_id", "unknown"), report)
    flash("已生成一份不含聊天原话的家庭关怀报告。", "success")
    return redirect(url_for("family_briefs"))
```
